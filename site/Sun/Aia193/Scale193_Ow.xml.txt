function MakeScale193o() {
  var D:=new DataScale;
  D.Name:='Scale193o';
  D.Interpolate:=true;
  D.AddNullRange('#808080');
  D.AddRange(-10,'#80000A');
  D.AddRange(0,'#000000');
  var F:=1.27,C,Prev;
  while (F<=4096) {
    C:=Fmt('#%.2x%.2x%.2x',fR(F),fG(F),fB(F));
    if(C!=Prev) D.AddRange(F,C);
    Prev:=C;
    F:=F*1.125;
  }
  D.AddRange(10000,'#ffffff');
  D.AddRange(-10000,'#000001');
  return D;
}

function fR(x) {
  return SClip( iif(x>=91, Max(0,Log((x-64)/20,10))^1.01*100+20 , (x/28)^3) );
}

function fG(x) {
  return SClip( iif( x>80, log(x/12,10)^1.7 * 58 - 36, ((x/20)^2) / 3 ) );
}

function fB_v1(x) {
  return iif(x>300, Max(log(x/100,90)-0.08,-10)*160-10, (x/118)^3 );

function fB(x) {
  return SClip( iif(x<1800, iif(x>300, Max(log(x/100,90)-0.08,-10)^1.08*190-10, (x/118)^3 ), (x/41.5)^1.2 ) );
}
function SClip(x) {
  // Gamma+Contrast:
  x:=Max(IntGamma(Contrast(Max(x,0),222,41),-133),0);
  //
  var C=240, W=40, S=0.27,
      M=1/(1+exp(0-(x-C)/W/S)); // sigmoid 0 before C-W, 1 after C+w
  return Min(x-x*M+255*M*1,255) // smooth clip to 255 with linear start...
}

  function IntGamma(x,G) {
    G:=(256-G)/256;
    return 255*Power((x/255),G);
  }
  function Contrast(x,Amount,Center) {
    // this is used in FMap_Contrast()
    return (x-Center)*Amount/255+x;
  }
