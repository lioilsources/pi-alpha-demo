function MakeScale171n() { //New160320
  // Scale for *_0171.fits, as downloaded from http://jsoc.stanford.edu/data/aia/synoptic/2010/05/13/H0000/AIA20100513_0000_0304.fits
  // typical value range 0..1300, maximum arround 6052, average 218 ...
  var D:=new DataScale;
  D.Name:='SDO171';
  D.Interpolate:=True;
  D.AddNullRange('#808080');
  D.AddRange(-10,'#08000A');
  D.AddRange(0,'#000000');
  var F:=1.27;
  while (F<=4096) {
    D.AddRange(F,Fmt('#%.2x%.2x%.2x',R(F),G(F),B(F)));
    F:=F*1.125; //F:=F*1.25;
  }
  D.AddRange(10000,'#FFFFFF');
  D.AddRange(-10000,'#000000');
  return D;
  
  function R(x) {
    return Bounds(IntGamma(Sqrt(x)*227/32,Rg),0,255);
  }
  function G(x) {
    return Bounds(IntGamma(Sqrt(x)*157/32,Gg),0,255);
  }
  function B(x) {
    //return Bounds(Sqrt(x)/3,0,255);
    //x210304: return Bounds(Sqrt(x),0,255);
    //??? return Bounds(x^(2/3),0,255);
    //??? return Bounds((x<1500)?sqrt(x):(exp(x/1024)*9-1),0,255);
    //??? return Bounds((x/4096)^4*255,0,255);
    //??? return Bounds( (x/4096)^3*320, 0,255);
    //??? Clip( (x/4096)^3*256 * (1+Exp(-x/256)) ,0,255)
    //??? return Bounds( Min( (x/4096)^4*256, 2*(250-125*(1+Exp(-x/4096))) ) , 0,255);
    //return Bounds( Min( (x/4096)^3*512, 3.14*(250-125*(1+Exp(-x/4096))) ) , 0,255);
    return Bounds( IntGamma( Max(Sqrt(x)*0.25, Min( (x/4096)^3*512, 3.14*(250-125*(1+Exp(-x/4096))) )) ,Bg) , 0,255 );
  }
}
